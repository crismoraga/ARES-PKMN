cmake_minimum_required(VERSION 3.16)
project(ares_mgba C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LIBMGBA_ONLY ON CACHE BOOL "Core only" FORCE)
set(M_CORE_GBA ON CACHE BOOL "GBA" FORCE)
set(M_CORE_GB OFF CACHE BOOL "GB" FORCE)
set(BUILD_STATIC ON CACHE BOOL "Static mGBA" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "No mGBA DLL" FORCE)
set(BUILD_QT OFF CACHE BOOL "No UI" FORCE)
set(BUILD_SDL OFF CACHE BOOL "No UI" FORCE)
set(BUILD_GL OFF CACHE BOOL "No GL" FORCE)
set(BUILD_GLES2 OFF CACHE BOOL "No GLES" FORCE)
set(BUILD_GLES3 OFF CACHE BOOL "No GLES" FORCE)
set(USE_DEBUGGERS OFF CACHE BOOL "No debugger" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "No scripting" FORCE)
set(USE_ZLIB OFF CACHE BOOL "No compression dependency" FORCE)
set(USE_PNG OFF CACHE BOOL "No image dependency" FORCE)
set(USE_SQLITE3 OFF CACHE BOOL "No database dependency" FORCE)
set(USE_FFMPEG OFF CACHE BOOL "No video dependency" FORCE)
set(USE_LZMA OFF CACHE BOOL "No archive dependency" FORCE)
set(ENABLE_VFS ON CACHE BOOL "ROM file access" FORCE)
add_subdirectory(vendor/mgba)
add_library(ares_mgba SHARED ares_mgba.c)
target_link_libraries(ares_mgba PRIVATE mgba)
if(MINGW)
    target_link_options(ares_mgba PRIVATE -static -static-libgcc)
endif()
get_target_property(MGBA_DEFINITIONS mgba COMPILE_DEFINITIONS)
target_compile_definitions(ares_mgba PRIVATE ${MGBA_DEFINITIONS})
set_target_properties(ares_mgba PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(CONFIG Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${CONFIG}" CONFIG_UPPER)
    set_target_properties(ares_mgba PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${CMAKE_BINARY_DIR}")
endforeach()
